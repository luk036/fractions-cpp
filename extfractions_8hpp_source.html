30 template <typename Mn> CONSTEXPR14 auto gcd_recur(const Mn& _m, const Mn& _n) -> Mn {

32 // With -O3 the compiler optimizes this into a jump (tail-call elimination),

65 template <typename Mn> CONSTEXPR14 auto lcm(const Mn& _m, const Mn& _n) -> Mn {
66 if (_m == 0 || _n == 0) {
67 return 0;
68 }
69 Mn a = _m < 0 ? -_m : _m;
70 Mn b = _n < 0 ? -_n : _n;
71 return (a / gcd(a, b)) * b;
72 }
