68 template <typename T> CONSTEXPR14 auto less_than_frac(T an, T ad, T bn, T bd) -> bool {
69# if defined(__SIZEOF_INT128__)
70 __extension__ typedef __int128 wide;
71 return static_cast<wide>(an) * static_cast<wide>(bd)
72 < static_cast<wide>(bn) * static_cast<wide>(ad);
73# elif defined(_MSC_VER) && defined(_M_X64)
74 const bool neg_a = is_negative(an);
75 const bool neg_b = is_negative(bn);
76 if (neg_a != neg_b) {
77 return neg_a;

80 unsigned long long hi_b = 0;

85 const int cmp = compare_u128(hi_a, lo_a, hi_b, lo_b);
86 return neg_a ? cmp > 0 : cmp < 0;
87# else

104 template <typename Mn> CONSTEXPR14 auto gcd_recur(const Mn& _m, const Mn& _n) -> Mn {
105 // DO NOT replace this tail recursion with a while loop.
106 // With -O3 the compiler optimizes this into a jump (tail-call elimination),
107 // making it faster than the iterative equivalent. This is intentional.
108 if (_n == 0) {
109 return _m < 0 ? -_m : _m;
110 }
111 return gcd_recur(_n, _m % _n);
112 }

139 template <typename Mn> CONSTEXPR14 auto lcm(const Mn& _m, const Mn& _n) -> Mn {
