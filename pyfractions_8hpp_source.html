845 // Handle sign for floor division
846 if ((numerator_product >= 0) == (denominator_product > 0)) {
847 // Same sign or numerator is 0 - simple division
848 return numerator_product / denominator_product;
849 }
850 // Different signs - need to adjust for floor
851 T quotient = numerator_product / denominator_product;
852 if (numerator_product % denominator_product != 0) {
853 return quotient - 1;
854 }
855 return quotient;
856 }

863 Fraction operator%(const Fraction& other) const {
864 if (other._numerator == 0) {
865 throw std::runtime_error("Modulo by zero");
866 }
867
868 T da = _denominator;
869 T db = other._denominator;
870
871 // Check for potential overflow in multiplications
872 bool overflow_risk = _mul_overflows(_numerator, db)
873 || _mul_overflows(other._numerator, da) || _mul_overflows(da, db);
874
875 if (overflow_risk) {
876 // Use floating point arithmetic to avoid overflow
877 double a = static_cast<double>(_numerator) / static_cast<double>(da);
878 double b = static_cast<double>(other._numerator) / static_cast<double>(db);
879 double result = std::fmod(a, b);
880
881 // Convert back to fraction
882 // Use a reasonable denominator based on original denominators
883 T new_denominator = da * db;
884 if (new_denominator < 0) new_denominator = -new_denominator;
885
886 T new_numerator
887 = static_cast<T>(std::round(result * static_cast<double>(new_denominator)));
888
889 // Normalize the result
890 T g = detail::gcd(new_numerator, new_denominator);
891 return Fraction(new_numerator / g, new_denominator / g);
892 }
893
894 // Safe to use integer arithmetic
895 T numerator_product = _numerator * db;
896 T denominator_product = other._numerator * da;
897 T n_mod = numerator_product % denominator_product;
898 T result_denominator = da * db;
899
900 return Fraction(n_mod, result_denominator);
901 }
